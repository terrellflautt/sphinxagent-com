Embed In Seconds

One Line of Code. That's It.

Add an AI chatbot to any website — WordPress, Shopify, React, or plain HTML. Or just share a link.

<!-- Paste before </body> -->
<script
  src="https://sphinxagent.ai/widget/sphinx-widget.js"
  data-agent-id="YOUR_AGENT_ID"
  data-color="#D4AF37">
</script>
// In your App.jsx or layout component
import { useEffect } from 'react';

function App() {
  useEffect(() => {
    const s = document.createElement('script');
    s.src = 'https://sphinxagent.ai/widget/sphinx-widget.js';
    s.dataset.agentId = 'YOUR_AGENT_ID';
    s.dataset.color = '#D4AF37';
    document.body.appendChild(s);
  }, []);
}
